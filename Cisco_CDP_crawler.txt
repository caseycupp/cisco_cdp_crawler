3.
Initialize Crawl State
Create data structures:
Queue (BFS crawl order)
Queued set (avoid duplicate enqueues)
Visited-OK set (devices successfully logged into)
Visited-Fail set (devices attempted but unreachable)
Rows list (CSV output)
Enqueue seed router (short hostname)


4.
Begin Crawl Loop
Repeat until queue is empty:

4.1 Dequeue Device
Get next device’s short hostname

4.2 Attempt SSH Login
Open SSH connection using Netmiko
Enter enable mode
Disable terminal paging
If SSH fails:
Mark hostname as unable_to_login
Record CSV row with:
site
hostname
status = unable_to_login
Add hostname to Visited-Fail
Skip neighbor discovery
Continue to next device

4.3 Collect Device Information

On successful login:
Run:
show version
show inventory
Parse and extract:
Hostname
Device type (router / switch)
Model
Serial number
Firmware version
Record CSV row with:
status = ok
Add hostname to Visited-OK

4.4 Discover Adjacent Devices (CDP)
Run:
show cdp neighbors detail
Parse neighbor entries:
Device ID / hostname
Normalize each neighbor hostname:
Strip domain
Lowercase

4.5 Neighbor Filtering & Enqueue
For each neighbor:
Apply naming convention filter (e.g. s\d+$)
Check if neighbor is already:
Visited-OK
Visited-Fail
Queued
If not seen yet:
Enqueue neighbor
Track it as queued
Do not attempt SSH immediately (queue controls flow)


5.
Crawl Completion
Queue is empty
All reachable and unreachable site devices identified


6.
Write CSV File
Open CSV using generated filename
Write header
Write one row per device:
One row per hostname
Includes unreachable devices


7.
Exit / Summary (Optional)
Print summary:
Total devices discovered
Successfully logged in
Unable to login
Exit cleanly

from netmiko import ConnectHandler
from collections import deque
import csv
import re
import getpass

# =========================
# CONFIGURATION
# =========================
DOMAIN = "williams.com"
DEVICE_TYPE = "cisco_ios"

SWITCH_REGEX = re.compile(r"s\d+$", re.IGNORECASE)

CSV_FIELDS = [
    "site",
    "hostname",
    "status",
    "device_type",
    "model",
    "serial_number",
    "firmware"
]

# =========================
# HELPER FUNCTIONS
# =========================
def short_hostname(name: str) -> str:
    return name.split(".")[0].lower().strip()

def fqdn_from_short(short: str) -> str:
    return f"{short}.{DOMAIN}"

def site_csv_filename(site: str) -> str:
    return f"{site.strip()[:5].lower()}.csv"

def connect_device(fqdn: str, username: str, password: str, secret: str):
    conn = ConnectHandler(
        device_type=DEVICE_TYPE,
        host=fqdn,
        username=username,
        password=password,
        secret=secret,
        fast_cli=False
    )
    conn.enable()
    conn.send_command("terminal length 0")
    return conn

# =========================
# PARSING FUNCTIONS (simple v1)
# =========================
def get_device_facts(conn):
    output = conn.send_command("show version")

    hostname = conn.find_prompt().replace("#", "").replace(">", "")
    hostname = short_hostname(hostname)

    model = ""
    serial = ""
    firmware = ""

    for line in output.splitlines():
        if "Cisco IOS XE Software" in line or "Cisco IOS Software" in line:
            firmware = line.strip()
        if "Model Number" in line:
            model = line.split(":")[-1].strip()
        if "Processor board ID" in line:
            serial = line.split()[-1].strip()

    device_type = "switch" if SWITCH_REGEX.search(hostname) else "router"

    return {
        "hostname": hostname,
        "device_type": device_type,
        "model": model,
        "serial_number": serial,
        "firmware": firmware
    }

def get_cdp_neighbors(conn):
    output = conn.send_command("show cdp neighbors detail")
    neighbors = []

    for block in output.split("Device ID:"):
        if block.strip() == "":
            continue
        first_line = block.strip().splitlines()[0]
        neighbor = short_hostname(first_line)
        neighbors.append(neighbor)

    return neighbors

# =========================
# MAIN
# =========================
def main():
    print("\n=== Cisco CDP Site Inventory v1 ===\n")

    seed_input = input("Enter seed router hostname (short or FQDN): ").strip()
    site_name = input("Enter site name: ").strip()

    username = input("SSH username: ").strip()
    password = getpass.getpass("SSH password: ")
    secret = getpass.getpass("Enable secret: ")

    seed_short = short_hostname(seed_input)
    csv_file = site_csv_filename(site_name)

    queue = deque()
    queued = set()
    visited_ok = set()
    visited_fail = set()
    rows = []

    queue.append(seed_short)
    queued.add(seed_short)

    while queue:
        hostname = queue.popleft()
        fqdn = fqdn_from_short(hostname)

        print(f"[+] Processing {hostname}")

        try:
            conn = connect_device(fqdn, username, password, secret)
            facts = get_device_facts(conn)
            neighbors = get_cdp_neighbors(conn)
            conn.disconnect()

            visited_ok.add(hostname)

            rows.append({
                "site": site_name,
                "hostname": facts["hostname"],
                "status": "ok",
                "device_type": facts["device_type"],
                "model": facts["model"],
                "serial_number": facts["serial_number"],
                "firmware": facts["firmware"]
            })

            for n in neighbors:
                if not SWITCH_REGEX.search(n):
                    continue
                if n in visited_ok or n in visited_fail or n in queued:
                    continue
                queue.append(n)
                queued.add(n)

        except Exception as e:
            print(f"[-] Unable to login to {hostname}: {e}")
            visited_fail.add(hostname)

            rows.append({
                "site": site_name,
                "hostname": hostname,
                "status": "unable_to_login",
                "device_type": "",
                "model": "",
                "serial_number": "",
                "firmware": ""
            })

    print(f"\n[+] Writing CSV: {csv_file}")

    with open(csv_file, "w", newline="") as f:
        writer = csv.DictWriter(f, fieldnames=CSV_FIELDS)
        writer.writeheader()
        for row in rows:
            writer.writerow(row)

    print("\n=== Done ===")
    print(f"Devices OK: {len(visited_ok)}")
    print(f"Unable to login: {len(visited_fail)}")

# =========================
if __name__ == "__main__":
    main()
